    <script>
        document.addEventListener("DOMContentLoaded", () => {
            const registrationForm = document.getElementById("registrationForm");
            const loginForm = document.getElementById("loginForm");
            const applicationForm = document.getElementById("applicationForm");

            const authSection = document.getElementById("authSection");
            const jobSection = document.getElementById("jobSection");
            const successSection = document.getElementById("successSection");
            const statusSection = document.getElementById("statusSection");

            const message = document.getElementById("message");
            const applicationIdSpan = document.getElementById("applicationId");
            const applicationStatus = document.getElementById("applicationStatus");

            let registeredUsers = [];
            let lastApplicationId = null;

            const statusOptions = [
                "Under Review",
                "Shortlisted",
                "Interview Scheduled",
                "Awaiting Documents",
                "In Progress",
                "On Hold",
                "Selected",
                "Rejected",
                "Position Closed",
                "Offer Sent"
            ];

            registrationForm.addEventListener("submit", (e) => {
                e.preventDefault();
                const email = document.getElementById("email").value;
                const password = document.getElementById("password").value;

                registeredUsers.push({ email, password });
                message.textContent = "Registered successfully! Please log in.";
                registrationForm.reset();
            });

            loginForm.addEventListener("submit", (e) => {
                e.preventDefault();
                const email = document.getElementById("loginEmail").value;
                const password = document.getElementById("loginPassword").value;

                const user = registeredUsers.find(u => u.email === email && u.password === password);

                if (user) {
                    authSection.style.display = "none";
                    jobSection.style.display = "block";
                } else {
                    alert("Invalid credentials. Please try again.");
                }
            });

            applicationForm.addEventListener("submit", (e) => {
                e.preventDefault();
                const appId = "APP" + Math.floor(100000 + Math.random() * 900000);
                const randomStatus = statusOptions[Math.floor(Math.random() * statusOptions.length)];

                lastApplicationId = appId;
                applicationIdSpan.textContent = appId;

                jobSection.style.display = "none";
                successSection.style.display = "block";
                applicationStatus.textContent = `Application ID ${appId} status: ${randomStatus}`;

                applicationForm.reset();
            });
        });
    </script>
